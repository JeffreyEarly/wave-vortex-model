cmake_minimum_required(VERSION 3.20)
project(WaveVortexKernelContractTests LANGUAGES CXX)

include(CTest)
add_subdirectory(${CMAKE_CURRENT_LIST_DIR}/../../CompiledKernel ${CMAKE_CURRENT_BINARY_DIR}/kernel)
add_subdirectory(${CMAKE_CURRENT_LIST_DIR}/../../PortableRuntime ${CMAKE_CURRENT_BINARY_DIR}/runtime)

add_executable(TestWVKernelContract tests/TestWVKernelContract.cpp tests/WVReferenceFFTEngine.cpp)
target_link_libraries(TestWVKernelContract PRIVATE WaveVortex::Kernel)
target_compile_features(TestWVKernelContract PRIVATE cxx_std_17)
set_target_properties(TestWVKernelContract PROPERTIES CXX_EXTENSIONS OFF)
if(CMAKE_CXX_COMPILER_ID MATCHES "AppleClang|Clang|GNU")
    target_compile_options(TestWVKernelContract PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME WVKernelContract COMMAND TestWVKernelContract)

add_executable(TestWVCheckpointReader ../portable-runtime/tests/TestWVCheckpointReader.cpp)
target_link_libraries(TestWVCheckpointReader PRIVATE WaveVortex::Checkpoint)
target_compile_definitions(TestWVCheckpointReader PRIVATE WV_CHECKPOINT_FIXTURE_DIR="${CMAKE_CURRENT_LIST_DIR}/../portable-runtime/fixtures")
target_compile_features(TestWVCheckpointReader PRIVATE cxx_std_17)
set_target_properties(TestWVCheckpointReader PROPERTIES CXX_EXTENSIONS OFF)
if(CMAKE_CXX_COMPILER_ID MATCHES "AppleClang|Clang|GNU")
    target_compile_options(TestWVCheckpointReader PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME WVCheckpointReader COMMAND TestWVCheckpointReader)
