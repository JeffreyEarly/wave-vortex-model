cmake_minimum_required(VERSION 3.20)
project(WVTiledScreen LANGUAGES CXX)
get_filename_component(WVM_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(FFTW_ROOT "" CACHE PATH "Frozen native FFTW provider")
find_package(Threads REQUIRED)
find_library(FFTW_LIBRARY fftw3 PATHS "${FFTW_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
find_library(FFTW_THREADS_LIBRARY fftw3_threads PATHS "${FFTW_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
add_executable(wv-tiled-screen TiledWorker.cpp "${WVM_ROOT}/CompiledKernel/src/WVKernelTypes.cpp")
target_compile_features(wv-tiled-screen PRIVATE cxx_std_17)
target_include_directories(wv-tiled-screen PRIVATE "${WVM_ROOT}/CompiledKernel/include" "${WVM_ROOT}/CompiledKernel/adapters/native-fftw" "${WVM_ROOT}/PortableRuntime/third_party" "${FFTW_ROOT}/include")
target_compile_options(wv-tiled-screen PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(wv-tiled-screen PRIVATE "${FFTW_THREADS_LIBRARY}" "${FFTW_LIBRARY}" Threads::Threads ${CMAKE_DL_LIBS})
set_target_properties(wv-tiled-screen PROPERTIES CXX_EXTENSIONS OFF BUILD_RPATH "${FFTW_ROOT}/lib")
